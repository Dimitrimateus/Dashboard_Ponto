Attribute VB_Name = "modPrepararAbasDoMes"
Option Explicit

' ESTE MÓDULO TEM DUAS MACROS (Alt+F8):
'   GerarCSVsDaPasta (02/11) - a do dia a dia: pede a pasta com os relatórios baixados
'       e a pasta de destino, traz tudo, formata, gera a aba CSV e grava os CSVs.
'       (descrita no fim do módulo)
'   PrepararAbasDoMes (31/10) - só formata as abas de relatório que você já colou na
'       planilha (sem gerar os CSVs).
'
' ================================================================================
' MACRO: PrepararAbasDoMes  (31/10)
' --------------------------------------------------------------------------------
' OBJETIVO
'   Formatar, de uma vez, TODOS os relatórios brutos do sistema que foram colados
'   como abas na planilha de Tratamento do mês, e deixar uma aba "Relatório da
'   preparação" dizendo, aba por aba, o que foi feito, se deu certo e o que deu
'   errado. Depois dela, é só rodar a GerarAbaCSV / ExportarCSVPorGestor.
'
' COMO USAR
'   1) Cole na planilha os módulos de todas as macros (cada .txt num módulo):
'      ConsolidarCartaoPonto, FormatarPausasTermicas, FormatarMarcacoesDigitadas,
'      FormatarAfastamentos, FormatarJornada e este. (O GerarCSVPonto também, para
'      o passo seguinte.)
'   2) Para cada relatório baixado do sistema, copie a aba dele para dentro da
'      planilha (botão direito na aba > Mover ou copiar > planilha de Tratamento,
'      marcando "Criar uma cópia"). O nome da aba não importa. O PontoNet (.csv)
'      também: abra o .csv no Excel e copie a aba.
'   3) Alt+F8 > PrepararAbasDoMes > Executar.
'
' COMO ELA RECONHECE CADA RELATÓRIO (pelo conteúdo, não pelo nome da aba)
'   Cartão Ponto (HRCP102) ......... "Cartão Ponto" + linhas "Empregado:"
'   Pausas térmicas (HRES114) ...... título com "Pausas Térmicas" (sem a tabela
'                                    já formatada "Pausa corretas" na linha 1)
'   Marcações digitadas ............ cabeçalhos "Crachá" e "Coletor"
'   Afastamentos (HRCL006) ......... "Histórico de Afastamentos" ou "Prev Término"
'   Jornada (intra, inter, inter
'   semanal, horas excedentes) ..... "Registros de Ponto", "Horas Descansadas",
'                                    "Horas DSR"/"Marcação Antes DSR" ou "Carga Hor."
'   PontoNet (Ausência de marcação)  A1 começa com "Data falta"
'   Aba que já está formatada, ou que não é relatório (RE, ORG, Tratamento...),
'   não é mexida.
'
' O QUE ACONTECE COM CADA UM
'   Cada relatório é formatado pela macro dele, no "modo automático" (as funções
'   <Macro>_Auto, que fazem o mesmo trabalho sem mostrar mensagens e devolvem o
'   texto da mensagem). Particularidades:
'   - Cartão Ponto: a macro dele cria a aba "Cartao_Consolidado"; aqui ela é
'     renomeada para "Cartão <Mês>" (pelo último dia do período) e a aba bruta é
'     apagada. Se já existe "Cartão <Mês>", a nova fica como "Cartão <Mês> (novo)"
'     e o relatório avisa.
'   - Pausas térmicas: com mais de um mês no mesmo relatório, sai uma aba por mês
'     (ver FormatarPausasTermicas).
'   - PontoNet: se o .csv veio todo na coluna A (separado por ";"), é separado em
'     colunas; a aba vira "Pontonet".
'   Um erro numa aba não para as outras: fica anotado no relatório.
'
' O RELATÓRIO (aba "Relatório da preparação", refeita a cada execução)
'   Aba original | O que era | Resultado | Aba(s) gerada(s) | Detalhes
'   e, embaixo, a lista do que se espera num mês completo, com o que FALTOU
'   (ex.: "Afastamentos: não encontrei nenhum relatório").
'
' Nunca rodou no Excel: rode numa CÓPIA da planilha na primeira vez.
' ================================================================================

Private Const NOME_RELATORIO As String = "Relatório da preparação"
' True quando a GerarCSVsDaPasta está rodando: a aba nova de um relatório SUBSTITUI a
' já formatada do mesmo mês/tipo (em vez de ficar como "(novo)").
Private substituirAbas As Boolean

Sub PrepararAbasDoMes()
    Dim wb As Workbook, abas As Collection, ws As Worksheet
    Dim linhas As Collection, vistos As Object
    Set wb = ThisWorkbook

    If MsgBox("Esta macro vai FORMATAR todas as abas de relatório do sistema que estão nesta planilha " & _
              "(Cartão Ponto, Pausas Térmicas, Marcações digitadas, Afastamentos, Jornada e PontoNet)." & vbCrLf & vbCrLf & _
              "As abas que já estão formatadas e as outras (RE, ORG, Tratamento...) não são mexidas." & vbCrLf & _
              "No fim, a aba """ & NOME_RELATORIO & """ mostra o que foi feito em cada uma." & vbCrLf & vbCrLf & _
              "Recomendado: rode numa CÓPIA da planilha na primeira vez. Continuar?", _
              vbYesNo + vbQuestion, "Preparar as abas do mês") <> vbYes Then Exit Sub

    ' guarda a lista de abas antes de começar (as macros criam, renomeiam e apagam abas)
    Set abas = New Collection
    For Each ws In wb.Worksheets
        If ws.Name <> NOME_RELATORIO Then abas.Add ws
    Next ws

    Set linhas = New Collection
    Set vistos = CreateObject("Scripting.Dictionary")
    Application.ScreenUpdating = False
    PrepararLista wb, abas, linhas, vistos, Nothing

    Application.StatusBar = False
    Application.Calculation = xlCalculationAutomatic
    Application.DisplayAlerts = True
    EscreverRelatorio wb, linhas, vistos
    Application.ScreenUpdating = True

    Dim nErros As Long, l As Variant
    For Each l In linhas
        If l(2) = "Deu erro" Then nErros = nErros + 1
    Next l
    MsgBox linhas.Count & " relatório(s) encontrado(s) e preparado(s)" & _
           IIf(nErros > 0, ", " & nErros & " com erro", ", todos OK") & "." & vbCrLf & vbCrLf & _
           "Veja os detalhes na aba """ & NOME_RELATORIO & """." & vbCrLf & _
           "Se estiver tudo certo, o próximo passo é a GerarAbaCSV.", _
           IIf(nErros > 0, vbExclamation, vbInformation), "Preparar as abas do mês"
End Sub

' --------------------------------------------------------------------------------
' Formata cada aba da lista que for relatório bruto e anota o resultado em
' "linhas" (para o relatório). Usada pela PrepararAbasDoMes (todas as abas) e pela
' GerarCSVsDaPasta (só as abas trazidas dos arquivos; origemPorAba = nome da aba
' -> nome do arquivo de onde ela veio, para o relatório mostrar o arquivo).
' --------------------------------------------------------------------------------
Private Sub PrepararLista(wb As Workbook, abas As Collection, linhas As Collection, vistos As Object, origemPorAba As Object)
    Dim ws As Worksheet, i As Long, tipo As String, resultado As String, detalhe As String, geradas As String
    For i = 1 To abas.Count
        Set ws = abas(i)
        tipo = TipoDoRelatorioBruto(ws)
        If tipo <> "" Then
            Application.StatusBar = "Preparando a aba " & ws.Name & " (" & NomeDoTipo(tipo) & ")..."
            vistos(tipo) = True
            Dim nomeOriginal As String
            nomeOriginal = ws.Name
            geradas = "": detalhe = ""
            Dim antes As Collection
            Set antes = FotoDasAbas(wb)
            On Error Resume Next
            Err.Clear
            Select Case tipo
                Case "cartao": detalhe = PrepararCartao(wb, ws, geradas)
                Case "pontonet": detalhe = PrepararPontoNet(wb, ws, geradas)
                Case Else
                    ws.Activate
                    detalhe = CStr(Application.Run(MacroDoTipo(tipo)))
                    If Err.Number = 0 Then geradas = AbasGeradas(wb, ws, antes)
            End Select
            If Err.Number <> 0 Then
                If Err.Number = 1004 And InStr(1, Err.Description, "macro", vbTextCompare) > 0 Then
                    detalhe = "ERRO: a macro " & MacroDoTipo(tipo) & " não está nesta planilha. Cole o módulo dela (o .txt) e rode de novo."
                Else
                    detalhe = "ERRO " & Err.Number & ": " & Err.Description & IIf(detalhe <> "", " (" & detalhe & ")", "")
                End If
            End If
            On Error GoTo 0
            If InStr(1, detalhe, "ERRO", vbBinaryCompare) > 0 Then resultado = "Deu erro" Else resultado = "OK"
            If Not origemPorAba Is Nothing Then
                If origemPorAba.Exists(nomeOriginal) Then nomeOriginal = "arquivo " & origemPorAba(nomeOriginal)
            End If
            linhas.Add Array(nomeOriginal, NomeDoTipo(tipo), resultado, geradas, detalhe)
        End If
    Next i

End Sub

' --------------------------------------------------------------------------------
' Que relatório bruto é esta aba ("" = não é relatório bruto, ou já está formatado).
' Olha o texto das primeiras 40 linhas x 34 colunas.
' --------------------------------------------------------------------------------
Private Function TipoDoRelatorioBruto(ws As Worksheet) As String
    Dim v As Variant, r As Long, c As Long, t As String, tudo As String, colA As String
    If Application.WorksheetFunction.CountA(ws.Cells) = 0 Then Exit Function
    v = ws.Range("A1:AH40").Value2
    For r = 1 To 40
        For c = 1 To 34
            If Not IsError(v(r, c)) Then
                t = UCase$(Trim$(Replace(CStr(v(r, c)), Chr(160), " ")))
                If t <> "" Then
                    tudo = tudo & "|" & t
                    If c = 1 Then colA = colA & "|" & t
                End If
            End If
        Next c
    Next r
    tudo = tudo & "|": colA = colA & "|"

    ' já formatadas (cabeçalho das macros na linha 1 ou 2): não mexe
    Dim l1 As String, l2 As String
    For c = 1 To 34
        If Not IsError(v(1, c)) Then l1 = l1 & "|" & UCase$(Trim$(CStr(v(1, c))))
        If Not IsError(v(2, c)) Then l2 = l2 & "|" & UCase$(Trim$(CStr(v(2, c))))
    Next c
    l1 = l1 & "|": l2 = l2 & "|"
    If InStr(l2, "|MATRÍCULA|") > 0 And InStr(l2, "|DT|") > 0 Then Exit Function          ' Cartão já consolidado
    If InStr(l1, "|PAUSA CORRETAS|") > 0 Then Exit Function                               ' pausas já formatadas
    If InStr(l1, "|MATRÍCULA|") > 0 And (InStr(l1, "|MOTIVO|") > 0 Or InStr(l1, "|SITUAÇÃO|") > 0 _
        Or InStr(l1, "|OCORRÊNCIA|") > 0 Or InStr(l1, "|INTERVALO TOTAL|") > 0 Or InStr(l1, "|TOTAL|") > 0) Then Exit Function

    If Left$(UCase$(Trim$(CStr(v(1, 1)))), 10) = "DATA FALTA" Then
        ' a aba "Pontonet" já separada em colunas é a formatada: não mexe
        If LCase$(ws.Name) = "pontonet" And InStr(CStr(v(1, 1)), ";") = 0 Then Exit Function
        TipoDoRelatorioBruto = "pontonet": Exit Function
    End If
    If InStr(tudo, "|CARTÃO PONTO|") > 0 And InStr(colA, "|EMPREGADO:|") > 0 Then TipoDoRelatorioBruto = "cartao": Exit Function
    If InStr(tudo, "PAUSAS TÉRMICAS") > 0 Then TipoDoRelatorioBruto = "pausas": Exit Function
    If InStr(tudo, "|CRACHÁ|") > 0 And InStr(tudo, "|COLETOR|") > 0 Then TipoDoRelatorioBruto = "digitadas": Exit Function
    If InStr(tudo, "HISTÓRICO DE AFASTAMENTOS") > 0 Or (InStr(tudo, "|PREV TÉRMINO|") > 0 And InStr(tudo, "|DT.TÉRMINO|") > 0) Then
        TipoDoRelatorioBruto = "afastamentos": Exit Function
    End If
    If InStr(tudo, "|REGISTROS DE PONTO|") > 0 Or InStr(tudo, "|HORAS DESCANSADAS|") > 0 Or InStr(tudo, "|HORAS DSR|") > 0 _
        Or InStr(tudo, "|MARCAÇÃO ANTES DSR|") > 0 Or InStr(tudo, "|CARGA HOR.|") > 0 Then
        TipoDoRelatorioBruto = "jornada": Exit Function
    End If
End Function

Private Function NomeDoTipo(ByVal tipo As String) As String
    Select Case tipo
        Case "cartao": NomeDoTipo = "Cartão Ponto"
        Case "pausas": NomeDoTipo = "Pausas Térmicas"
        Case "digitadas": NomeDoTipo = "Marcações digitadas"
        Case "afastamentos": NomeDoTipo = "Afastamentos"
        Case "jornada": NomeDoTipo = "Jornada (intra, inter, semanal ou horas excedentes)"
        Case "pontonet": NomeDoTipo = "PontoNet (ausência de marcação)"
    End Select
End Function

Private Function MacroDoTipo(ByVal tipo As String) As String
    Select Case tipo
        Case "cartao": MacroDoTipo = "ConsolidarCartaoPontoDefinitivo_Auto"
        Case "pausas": MacroDoTipo = "FormatarPausasTermicas_Auto"
        Case "digitadas": MacroDoTipo = "FormatarMarcacoesDigitadas_Auto"
        Case "afastamentos": MacroDoTipo = "FormatarAfastamentos_Auto"
        Case "jornada": MacroDoTipo = "FormatarJornada_Auto"
        Case "pontonet": MacroDoTipo = "(nenhuma: feito por esta macro)"
    End Select
End Function

' As abas que existem agora (para saber, depois de uma formatação, quais são novas).
Private Function FotoDasAbas(wb As Workbook) As Collection
    Dim sh As Worksheet
    Set FotoDasAbas = New Collection
    For Each sh In wb.Worksheets
        FotoDasAbas.Add sh
    Next sh
End Function

' Nomes das abas que saíram da formatação: a própria aba (que pode ter sido
' renomeada) e as abas novas que não existiam antes (ex.: pausas de vários meses).
Private Function AbasGeradas(wb As Workbook, ws As Worksheet, abasAntes As Collection) As String
    Dim sh As Worksheet, a As Variant, nova As Boolean, t As String
    t = ws.Name
    For Each sh In wb.Worksheets
        If sh.Name <> NOME_RELATORIO And sh.Name <> ws.Name Then
            nova = True
            For Each a In abasAntes
                If a Is sh Then nova = False: Exit For
            Next a
            If nova And InStr(1, "|" & t & "|", "|" & sh.Name & "|") = 0 Then t = t & ", " & sh.Name
        End If
    Next sh
    AbasGeradas = t
End Function

' --------------------------------------------------------------------------------
' Cartão Ponto: roda a ConsolidarCartaoPonto (que grava em "Cartao_Consolidado"),
' renomeia o resultado para "Cartão <Mês>" e apaga a aba bruta.
' --------------------------------------------------------------------------------
Private Function PrepararCartao(wb As Workbook, ws As Worksheet, ByRef geradas As String) As String
    Dim msg As String, wsC As Worksheet, ult As Long, r As Long, maxDt As Date, nome As String
    ' se sobrou um "Cartao_Consolidado" de antes, tira do caminho (a macro o apagaria)
    If AbaExisteAqui(wb, "Cartao_Consolidado") Then wb.Worksheets("Cartao_Consolidado").Name = "Cartao_Consolidado " & Format(Now, "hhnnss")
    ws.Activate
    msg = CStr(Application.Run("ConsolidarCartaoPontoDefinitivo_Auto"))
    If Not AbaExisteAqui(wb, "Cartao_Consolidado") Then
        PrepararCartao = "ERRO: a consolidação não gerou a aba. " & msg
        Exit Function
    End If
    Set wsC = wb.Worksheets("Cartao_Consolidado")
    If InStr(msg, "ERRO") > 0 Then
        Application.DisplayAlerts = False
        wsC.Delete
        Application.DisplayAlerts = True
        PrepararCartao = msg
        Exit Function
    End If
    ult = wsC.Cells(wsC.Rows.Count, 5).End(xlUp).Row
    For r = 3 To ult
        If IsDate(wsC.Cells(r, 5).Value) Then
            If wsC.Cells(r, 5).Value > maxDt Then maxDt = wsC.Cells(r, 5).Value
        End If
    Next r
    If maxDt = 0 Then
        PrepararCartao = "ERRO: não achei as datas na aba consolidada. " & msg
        Exit Function
    End If
    nome = "Cartão " & NomeDoMesAqui(Month(maxDt))
    If AbaExisteAqui(wb, nome) And substituirAbas Then
        Application.DisplayAlerts = False
        wb.Worksheets(nome).Delete
        Application.DisplayAlerts = True
        msg = msg & " Substituiu a aba """ & nome & """ que já existia."
    End If
    If AbaExisteAqui(wb, nome) Then
        msg = msg & " ATENÇÃO: já existia uma aba """ & nome & """; a nova ficou como """ & nome & " (novo)"". " & _
              "Apague a que estiver errada (o GerarCSV usa as 3 abas de Cartão mais recentes)."
        nome = nome & " (novo)"
    End If
    wsC.Name = nome
    geradas = nome
    Application.DisplayAlerts = False
    ws.Delete     ' a aba bruta: o consolidado já tem tudo
    Application.DisplayAlerts = True
    PrepararCartao = msg & " A aba bruta foi apagada."
End Function

' --------------------------------------------------------------------------------
' PontoNet ("Ausência de marcação", .csv): se veio tudo na coluna A, separa pelo
' ";" (datas no formato dia/mês/ano; "Código terceiro" como texto) e renomeia a
' aba para "Pontonet" (o nome que o GerarCSV procura).
' --------------------------------------------------------------------------------
Private Function PrepararPontoNet(wb As Workbook, ws As Worksheet, ByRef geradas As String) As String
    Dim msg As String, ult As Long
    ult = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    If InStr(CStr(ws.Cells(1, 1).Value), ";") > 0 Then
        ws.Range("A1:A" & ult).TextToColumns Destination:=ws.Range("A1"), DataType:=xlDelimited, _
            TextQualifier:=xlDoubleQuote, ConsecutiveDelimiter:=False, Tab:=False, Semicolon:=True, _
            Comma:=False, Space:=False, Other:=False, _
            FieldInfo:=Array(Array(1, xlDMYFormat), Array(11, xlTextFormat), Array(15, xlDMYFormat), _
                             Array(16, xlDMYFormat), Array(17, xlDMYFormat), Array(18, xlDMYFormat))
        msg = "O arquivo veio numa coluna só: separado em colunas pelo "";"". "
    End If
    If ws.Cells(1, 1).Value <> "Data falta" Then
        PrepararPontoNet = "ERRO: depois de separar, a coluna A não começa com ""Data falta"". Confira o arquivo."
        Exit Function
    End If
    ws.Rows(1).Font.Bold = True
    ws.Columns.AutoFit
    If LCase$(ws.Name) <> "pontonet" Then
        If AbaExisteAqui(wb, "Pontonet") Then
            msg = msg & "ATENÇÃO: já existe uma aba ""Pontonet"" e o GerarCSV usa ela; esta ficou como """ & ws.Name & """. Apague a antiga e renomeie esta."
        Else
            ws.Name = "Pontonet"
        End If
    End If
    geradas = ws.Name
    PrepararPontoNet = msg & (ult - 1) & " pedido(s) de justificativa."
End Function

' --------------------------------------------------------------------------------
' Aba "Relatório da preparação": uma linha por relatório encontrado e, embaixo, o
' que se espera num mês completo e se veio.
' --------------------------------------------------------------------------------
Private Sub EscreverRelatorio(wb As Workbook, linhas As Collection, vistos As Object)
    Dim ws As Worksheet, r As Long, l As Variant, esperados As Variant, i As Long
    If AbaExisteAqui(wb, NOME_RELATORIO) Then
        Set ws = wb.Worksheets(NOME_RELATORIO)
        ws.Cells.Clear
    Else
        Set ws = wb.Worksheets.Add(Before:=wb.Worksheets(1))
        ws.Name = NOME_RELATORIO
    End If
    ws.Range("A1").Value = "Preparação das abas do mês - " & Format(Now, "dd/mm/yyyy hh:nn")
    ws.Range("A1").Font.Bold = True
    ws.Range("A1").Font.Size = 13
    ws.Range("A3:E3").Value = Array("Aba original", "O que era", "Resultado", "Aba(s) gerada(s)", "Detalhes")
    ws.Range("A3:E3").Font.Bold = True
    ws.Range("A3:E3").Interior.Color = RGB(217, 217, 217)
    r = 4
    If linhas.Count = 0 Then
        ws.Cells(r, 1).Value = "Nenhum relatório bruto encontrado nas abas desta planilha."
        r = r + 1
    End If
    For Each l In linhas
        ws.Cells(r, 1).Value = l(0)
        ws.Cells(r, 2).Value = l(1)
        ws.Cells(r, 3).Value = l(2)
        ws.Cells(r, 4).Value = l(3)
        ws.Cells(r, 5).Value = l(4)
        ws.Cells(r, 3).Font.Bold = True
        ws.Cells(r, 3).Font.Color = IIf(l(2) = "OK", RGB(0, 120, 60), IIf(l(2) = "Deu erro", RGB(200, 0, 0), RGB(110, 110, 110)))
        r = r + 1
    Next l

    r = r + 1
    ws.Cells(r, 1).Value = "O que se espera num mês completo"
    ws.Cells(r, 1).Font.Bold = True
    r = r + 1
    esperados = Array("cartao", "pausas", "digitadas", "afastamentos", "jornada", "pontonet")
    For i = 0 To UBound(esperados)
        ws.Cells(r, 1).Value = NomeDoTipo(CStr(esperados(i)))
        If vistos.Exists(esperados(i)) Then
            ws.Cells(r, 3).Value = "Veio"
            ws.Cells(r, 3).Font.Color = RGB(0, 120, 60)
        ElseIf ExisteAbaFormatada(wb, CStr(esperados(i))) Then
            ws.Cells(r, 3).Value = "Já estava formatada"
            ws.Cells(r, 3).Font.Color = RGB(0, 120, 60)
        Else
            ws.Cells(r, 3).Value = "FALTOU: não encontrei nenhum relatório"
            ws.Cells(r, 3).Font.Color = RGB(200, 0, 0)
        End If
        r = r + 1
    Next i
    ws.Cells(r, 1).Value = "Também precisam estar na planilha: a RE, o ORG e as abas Tratamento - Internos/Externos (essas não são mexidas)."

    ws.Columns("A").ColumnWidth = 28
    ws.Columns("B").ColumnWidth = 30
    ws.Columns("C").ColumnWidth = 16
    ws.Columns("D").ColumnWidth = 34
    ws.Columns("E").ColumnWidth = 90
    ws.Range("E:E").WrapText = True
    ws.Range("A:E").VerticalAlignment = xlTop
    ws.Activate
    ws.Range("A1").Select
End Sub

' Para a lista do que se espera: já existe uma aba formatada daquele tipo?
Private Function ExisteAbaFormatada(wb As Workbook, ByVal tipo As String) As Boolean
    Dim sh As Worksheet, n As String
    For Each sh In wb.Worksheets
        n = LCase$(sh.Name)
        Select Case tipo
            Case "cartao": If Left$(n, 6) = "cartão" Or Left$(n, 6) = "cartao" Then ExisteAbaFormatada = True
            Case "pausas": If Left$(n, 15) = "pausas térmicas" Then ExisteAbaFormatada = True
            Case "digitadas": If n = "marcações digitadas" Then ExisteAbaFormatada = True
            Case "afastamentos": If n = "afastamentos" Then ExisteAbaFormatada = True
            Case "jornada": If n = "intrajornada" Or n = "interjornada" Or n = "interjornada semanal" Or n = "horas excedentes" Then ExisteAbaFormatada = True
            Case "pontonet": If n = "pontonet" Then ExisteAbaFormatada = True
        End Select
    Next sh
End Function

Private Function AbaExisteAqui(wb As Workbook, ByVal nome As String) As Boolean
    Dim sh As Object
    For Each sh In wb.Sheets
        If LCase$(sh.Name) = LCase$(nome) Then AbaExisteAqui = True: Exit Function
    Next sh
End Function

Private Function NomeDoMesAqui(ByVal m As Long) As String
    NomeDoMesAqui = Choose(m, "Janeiro", "Fevereiro", "Março", "Abril", "Maio", "Junho", "Julho", _
                           "Agosto", "Setembro", "Outubro", "Novembro", "Dezembro")
End Function

' ================================================================================
' MACRO: GerarCSVsDaPasta  (02/11)
' --------------------------------------------------------------------------------
' Faz o mês inteiro de uma vez, a partir de uma PASTA com os relatórios baixados:
'   1) pede a pasta onde estão os arquivos do sistema (Cartão Ponto de cada mês,
'      Pausas Térmicas, Marcações digitadas, Afastamentos, os 4 de jornada, o
'      PontoNet .csv e, se quiser, a RE nova);
'   2) pede a pasta onde os CSVs vão ser gravados;
'   3) abre cada arquivo, traz as abas dele para esta planilha e fecha (os arquivos
'      da pasta não são alterados);
'   4) formata tudo (a mesma preparação da PrepararAbasDoMes). Uma aba já formatada
'      do mesmo relatório é SUBSTITUÍDA pela nova (ex.: "Marcações Digitadas",
'      "Pausas Térmicas Outubro", "Cartão Outubro"), para rodar de novo no mês sem
'      duplicar nada;
'   5) roda a GerarAbaCSV e grava os CSVs (dados_<gestor>.csv + dados_TODOS.csv) na
'      pasta escolhida;
'   6) deixa a aba "Relatório da preparação" com tudo o que aconteceu.
' Ficam na planilha (não vêm da pasta): ORG e as abas Tratamento - Internos/Externos.
' A RE pode vir na pasta: entra como "RE dd.mm" (data do arquivo) ANTES das REs que
' já existem, e o GerarCSV passa a usar a nova (as antigas não são apagadas, porque
' fórmulas da planilha podem apontar para elas).
' Arquivo que não é nenhum desses relatórios é ignorado (e aparece no relatório).
' Precisa dos módulos de todas as macros colados na planilha, inclusive o GerarCSVPonto.
' ================================================================================
Sub GerarCSVsDaPasta()
    Dim wb As Workbook, pastaOrigem As String, pastaDestino As String
    Dim arquivos As Collection, nomeArq As String, ext As String, f As Variant
    Set wb = ThisWorkbook

    pastaOrigem = EscolherPastaAqui("1 de 2: escolha a PASTA com os relatórios baixados do sistema")
    If pastaOrigem = "" Then Exit Sub
    pastaDestino = EscolherPastaAqui("2 de 2: escolha a pasta onde os arquivos CSV vão ser gravados")
    If pastaDestino = "" Then Exit Sub

    ' arquivos de planilha e .csv da pasta (sem os temporários "~$" e sem esta planilha)
    Set arquivos = New Collection
    nomeArq = Dir(pastaOrigem & "\*.*")
    Do While nomeArq <> ""
        ext = LCase$(Mid$(nomeArq, InStrRev(nomeArq, ".") + 1))
        If (ext = "xlsx" Or ext = "xls" Or ext = "xlsm" Or ext = "csv") And Left$(nomeArq, 2) <> "~$" _
           And LCase$(pastaOrigem & "\" & nomeArq) <> LCase$(wb.FullName) Then arquivos.Add nomeArq
        nomeArq = Dir()
    Loop
    If arquivos.Count = 0 Then
        MsgBox "Não encontrei nenhum arquivo .xlsx, .xls ou .csv na pasta:" & vbCrLf & pastaOrigem, vbExclamation, "Gerar CSVs da pasta"
        Exit Sub
    End If

    Dim lista As String
    For Each f In arquivos
        lista = lista & vbCrLf & "  - " & f
    Next f
    If MsgBox("Encontrei " & arquivos.Count & " arquivo(s):" & lista & vbCrLf & vbCrLf & _
              "Vou trazer os relatórios para esta planilha, formatar (substituindo as abas já formatadas " & _
              "do mesmo relatório), gerar a aba CSV e gravar os CSVs em:" & vbCrLf & pastaDestino & vbCrLf & vbCrLf & _
              "Recomendado: rode numa CÓPIA da planilha na primeira vez. Continuar?", _
              vbYesNo + vbQuestion, "Gerar CSVs da pasta") <> vbYes Then Exit Sub

    Dim linhas As Collection, vistos As Object, origemPorAba As Object, importadas As Collection
    Set linhas = New Collection
    Set vistos = CreateObject("Scripting.Dictionary")
    Set origemPorAba = CreateObject("Scripting.Dictionary")
    Set importadas = New Collection
    Application.ScreenUpdating = False

    ' 3) traz as abas de cada arquivo
    For Each f In arquivos
        Application.StatusBar = "Abrindo " & f & "..."
        ImportarArquivo wb, pastaOrigem & "\" & f, CStr(f), importadas, origemPorAba, linhas
    Next f

    ' RE nova e arquivos que não são relatório; o resto vai para a preparação
    ' (For Each numa Collection precisa de Variant; ws As Worksheet é o que as funções
    ' recebem: passar o Variant direto daria "ByRef argument type mismatch")
    Dim paraPreparar As Collection, elem As Variant, ws As Worksheet, tipo As String
    Set paraPreparar = New Collection
    For Each elem In importadas
        Set ws = elem
        tipo = TipoDoRelatorioBruto(ws)
        If tipo <> "" Then
            paraPreparar.Add ws
        ElseIf EhRE(ws) Then
            linhas.Add Array("arquivo " & origemPorAba(ws.Name), "RE (cadastro)", "OK", PrepararRE(wb, ws, CStr(origemPorAba(ws.Name))), _
                "Entrou antes das outras REs: o GerarCSV usa esta.")
        Else
            linhas.Add Array("arquivo " & origemPorAba(ws.Name) & " (aba " & ws.Name & ")", "Não reconhecido", "Ignorado", "", _
                "Não é nenhum dos relatórios esperados; a aba foi descartada.")
            Application.DisplayAlerts = False
            ws.Delete
            Application.DisplayAlerts = True
        End If
    Next elem

    ' 4) substitui as abas já formatadas dos mesmos relatórios e formata
    Dim msgSubst As String
    msgSubst = ApagarAbasQueSeraoSubstituidas(wb, paraPreparar)
    substituirAbas = True
    PrepararLista wb, paraPreparar, linhas, vistos, origemPorAba
    substituirAbas = False

    ' 5) aba CSV e arquivos CSV
    Dim nErros As Long, l As Variant, resCSV As String, resExp As String, seguir As Boolean
    For Each l In linhas
        If l(2) = "Deu erro" Then nErros = nErros + 1
    Next l
    seguir = True
    If nErros > 0 Then
        Application.ScreenUpdating = True
        seguir = (MsgBox(nErros & " relatório(s) deram erro na formatação (detalhes na aba """ & NOME_RELATORIO & """)." & vbCrLf & vbCrLf & _
                         "Gerar os CSVs mesmo assim?", vbYesNo + vbExclamation, "Gerar CSVs da pasta") = vbYes)
        Application.ScreenUpdating = False
    End If
    If seguir Then
        Application.StatusBar = "Gerando a aba CSV..."
        resCSV = RodarEtapaCSV("GerarAbaCSV_Auto", "")
        linhas.Add Array("-", "Gerar a aba CSV", IIf(InStr(resCSV, "ERRO") > 0, "Deu erro", "OK"), "CSV", resCSV)
        If InStr(resCSV, "ERRO") = 0 Then
            Application.StatusBar = "Gravando os arquivos CSV..."
            resExp = RodarEtapaCSV("ExportarCSVPorGestor_NaPasta", pastaDestino)
            linhas.Add Array("-", "Gravar os arquivos CSV", IIf(InStr(resExp, "ERRO") > 0, "Deu erro", "OK"), pastaDestino, resExp)
        End If
    Else
        linhas.Add Array("-", "Gerar os CSVs", "Não feito", "", "Você escolheu não gerar por causa dos erros. Corrija e rode de novo.")
    End If
    If msgSubst <> "" Then linhas.Add Array("-", "Abas substituídas", "OK", "", msgSubst)

    Application.StatusBar = False
    Application.Calculation = xlCalculationAutomatic
    Application.DisplayAlerts = True
    EscreverRelatorio wb, linhas, vistos
    Application.ScreenUpdating = True

    nErros = 0
    For Each l In linhas
        If l(2) = "Deu erro" Then nErros = nErros + 1
    Next l
    MsgBox IIf(nErros = 0 And seguir, "Pronto! Os CSVs foram gravados em:" & vbCrLf & pastaDestino, _
               "Terminou com " & nErros & " erro(s).") & vbCrLf & vbCrLf & _
           "Veja o que foi feito em cada arquivo na aba """ & NOME_RELATORIO & """.", _
           IIf(nErros > 0, vbExclamation, vbInformation), "Gerar CSVs da pasta"
End Sub

' Roda uma etapa do GerarCSVPonto (pelo nome) e devolve o texto dela; se o módulo
' não estiver colado, devolve o erro em palavras.
Private Function RodarEtapaCSV(ByVal macro As String, ByVal pasta As String) As String
    On Error Resume Next
    If pasta = "" Then
        RodarEtapaCSV = CStr(Application.Run(macro))
    Else
        RodarEtapaCSV = CStr(Application.Run(macro, pasta))
    End If
    If Err.Number <> 0 Then
        If Err.Number = 1004 And InStr(1, Err.Description, "macro", vbTextCompare) > 0 Then
            RodarEtapaCSV = "ERRO: a macro " & macro & " não está nesta planilha. Cole o módulo GerarCSVPonto (o .txt novo) e rode de novo."
        Else
            RodarEtapaCSV = "ERRO " & Err.Number & ": " & Err.Description & " " & RodarEtapaCSV
        End If
    End If
    On Error GoTo 0
End Function

Private Function EscolherPastaAqui(ByVal titulo As String) As String
    Dim fd As Object
    Set fd = Application.FileDialog(4)    ' 4 = escolher pasta
    fd.Title = titulo
    fd.AllowMultiSelect = False
    If fd.Show = -1 Then EscolherPastaAqui = fd.SelectedItems(1)
End Function

' Abre um arquivo (só leitura), copia para esta planilha cada aba dele que tenha algo
' escrito ("Importado 1", "Importado 2"...) e fecha sem salvar. O .csv abre com o
' separador e as datas do Windows em português (Local:=True).
Private Sub ImportarArquivo(wb As Workbook, ByVal caminho As String, ByVal nomeArq As String, _
                            importadas As Collection, origemPorAba As Object, linhas As Collection)
    Dim src As Workbook, sh As Worksheet, nova As Worksheet, n As Long
    On Error Resume Next
    Application.DisplayAlerts = False
    Set src = Workbooks.Open(Filename:=caminho, UpdateLinks:=0, ReadOnly:=True, Local:=True)
    Application.DisplayAlerts = True
    If src Is Nothing Or Err.Number <> 0 Then
        linhas.Add Array("arquivo " & nomeArq, "Não abriu", "Deu erro", "", "ERRO: não consegui abrir o arquivo (" & Err.Description & ").")
        Err.Clear
        On Error GoTo 0
        Exit Sub
    End If
    On Error GoTo 0
    For Each sh In src.Worksheets
        If Application.WorksheetFunction.CountA(sh.Cells) > 0 Then
            sh.Copy After:=wb.Sheets(wb.Sheets.Count)
            Set nova = wb.Sheets(wb.Sheets.Count)
            n = 1
            Do While AbaExisteAqui(wb, "Importado " & n)
                n = n + 1
            Loop
            nova.Name = "Importado " & n
            importadas.Add nova
            origemPorAba(nova.Name) = nomeArq
        End If
    Next sh
    src.Close SaveChanges:=False
End Sub

' A aba é a RE (cadastro)? Linha 1 com Matrícula (com ou sem acento) e "Nome Unidade".
Private Function EhRE(ws As Worksheet) As Boolean
    Dim c As Long, t As String, temMat As Boolean, temUni As Boolean
    For c = 1 To 40
        t = UCase$(Trim$(CStr(ws.Cells(1, c).Text)))
        If t = "MATRICULA" Or t = "MATRÍCULA" Then temMat = True
        If t = "NOME UNIDADE" Then temUni = True
    Next c
    EhRE = temMat And temUni
End Function

' RE trazida da pasta: nome "RE dd.mm" (data do arquivo) e posição antes da primeira
' RE que já existe (o GerarCSV usa a primeira RE da esquerda para a direita).
Private Function PrepararRE(wb As Workbook, ws As Worksheet, ByVal nomeArq As String) As String
    Dim nome As String, sh As Worksheet, k As Long
    nome = "RE " & Format(Date, "dd.mm")
    k = 2
    Do While AbaExisteAqui(wb, nome)
        nome = "RE " & Format(Date, "dd.mm") & " (" & k & ")"
        k = k + 1
    Loop
    ws.Name = nome
    For Each sh In wb.Worksheets
        If UCase$(Left$(sh.Name, 3)) = "RE " And Not sh Is ws Then
            ws.Move Before:=sh
            Exit For
        End If
    Next sh
    PrepararRE = nome
End Function

' Apaga as abas já formatadas que os relatórios trazidos vão substituir: Marcações
' Digitadas, Afastamentos, Pontonet, a aba de jornada do mesmo tipo e as Pausas
' Térmicas dos meses do relatório. (O Cartão é substituído na hora de renomear, em
' PrepararCartao, porque o mês só se sabe depois de consolidar.) Devolve a lista.
Private Function ApagarAbasQueSeraoSubstituidas(wb As Workbook, paraPreparar As Collection) As String
    Dim elem As Variant, ws As Worksheet, alvos As Collection, a As Variant, t As String, m As Variant
    Set alvos = New Collection
    For Each elem In paraPreparar
        Set ws = elem
        Select Case TipoDoRelatorioBruto(ws)
            Case "digitadas": alvos.Add "Marcações Digitadas"
            Case "afastamentos": alvos.Add "Afastamentos"
            Case "pontonet": alvos.Add "Pontonet"
            Case "jornada": If SubtipoJornada(ws) <> "" Then alvos.Add SubtipoJornada(ws)
            Case "pausas"
                For Each m In MesesDoRelatorioPausas(ws)
                    alvos.Add "Pausas Térmicas " & m
                Next m
        End Select
    Next elem
    Application.DisplayAlerts = False
    For Each a In alvos
        If AbaExisteAqui(wb, CStr(a)) And Not EstaNaLista(wb.Worksheets(CStr(a)), paraPreparar) Then
            wb.Worksheets(CStr(a)).Delete
            t = t & IIf(t = "", "", ", ") & a
        End If
    Next a
    Application.DisplayAlerts = True
    ApagarAbasQueSeraoSubstituidas = t
End Function

Private Function EstaNaLista(sh As Worksheet, lista As Collection) As Boolean
    Dim x As Variant
    For Each x In lista
        If x Is sh Then EstaNaLista = True: Exit Function
    Next x
End Function

' Nome da aba que a FormatarJornada vai dar a este relatório (mesmos títulos que ela olha).
Private Function SubtipoJornada(ws As Worksheet) As String
    Dim v As Variant, r As Long, c As Long, tudo As String
    v = ws.Range("A1:AH12").Value2
    For r = 1 To 12
        For c = 1 To 34
            If VarType(v(r, c)) = vbString Then tudo = tudo & " | " & UCase$(v(r, c))
        Next c
    Next r
    If InStr(tudo, "REGISTROS DE PONTO") > 0 And InStr(tudo, "INTERVALO") > 0 Then
        SubtipoJornada = "Intrajornada"
    ElseIf InStr(tudo, "HORAS DESCANSADAS") > 0 Then
        SubtipoJornada = "Interjornada"
    ElseIf InStr(tudo, "HORAS DSR") > 0 Or InStr(tudo, "MARCAÇÃO ANTES DSR") > 0 Or InStr(tudo, "MARCACAO ANTES DSR") > 0 Then
        SubtipoJornada = "Interjornada Semanal"
    ElseIf InStr(tudo, "CARGA HOR") > 0 And InStr(tudo, "EXTRAS") > 0 Then
        SubtipoJornada = "Horas Excedentes"
    End If
End Function

' Meses (ciclos de 28 a 27, pelo dia 27 que fecha cada um) do período do relatório de
' pausas, lidos na linha "Período:" (as datas dessa linha).
Private Function MesesDoRelatorioPausas(ws As Worksheet) As Collection
    Dim r As Long, c As Long, v As Variant, ini As Double, fim As Double, d As Date, f As Date
    Set MesesDoRelatorioPausas = New Collection
    For r = 1 To 15
        For c = 1 To 20
            If Trim$(CStr(ws.Cells(r, c).Text)) = "Período:" Then
                For c = 1 To 30
                    v = ws.Cells(r, c).Value2
                    If VarType(v) = vbDouble Then
                        If v > 40000 And v < 80000 Then
                            If ini = 0 Or v < ini Then ini = v
                            If v > fim Then fim = v
                        End If
                    End If
                Next c
                Exit For
            End If
        Next c
        If ini > 0 Then Exit For
    Next r
    If ini = 0 Then Exit Function
    d = CDate(ini)
    Do
        If Day(d) >= 28 Then f = DateSerial(Year(d), Month(d) + 1, 27) Else f = DateSerial(Year(d), Month(d), 27)
        MesesDoRelatorioPausas.Add NomeDoMesAqui(Month(f))
        d = f + 1
    Loop While d <= CDate(fim)
End Function
