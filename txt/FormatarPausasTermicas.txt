Attribute VB_Name = "modFormatarPausas"
Option Explicit

' Modo automático (31/10): quando a macro PrepararAbasDoMes chama esta formatação
' (FormatarPausasTermicas_Auto), nenhuma mensagem aparece; o texto de cada aviso
' vai para msgAuto e volta para o relatório da preparação. Ver Aviso, no fim.
Private modoAuto As Boolean
Private msgAuto As String

' ============================================================================
' FormatarPausasTermicas
'
' DE ONDE VEM: a aba ATIVA no momento em que a macro é executada (ws =
'   ActiveSheet). Espera-se que seja o "Relatório Analítico de Pausas
'   Térmicas" bruto, exportado como um único "dump" de impressão: cabeçalhos
'   de página repetidos a cada colaborador, uma tabela de marcações por dia,
'   um bloco "Total do Colaborador" ao final de cada colaborador, e um bloco
'   "Total Geral" (com a legenda e o rodapé do relatório) no final do
'   arquivo inteiro.
'
' PARA ONDE VAI E O QUE ESTE MACRO FAZ: reescreve essa MESMA aba (in-place),
'   substituindo o "dump" bruto por TRÊS tabelas limpas, uma abaixo da outra
'   (sem título de página, legenda, rodapé nem "Soma dos itens"):
'     1) Tabela de colaboradores - começa na linha 1 (cabeçalho) e tem uma
'        linha por colaborador, com os 7 números que o próprio relatório
'        original já calcula (Pausa corretas/menor/maior 0:20, Trabalho
'        correto/maior/menor 1:40, Marcações Ímpares).
'     2) Tabela de total - depois de UMA linha em branco, com o "Total
'        Geral" desses mesmos 7 números, também já calculado pelo relatório.
'     3) Marcações do período (18/10, pedido do RH) - depois de duas linhas em
'        branco: uma linha por batida (matrícula, colaborador, data, dia da
'        semana, hora, nº da batida no dia, tipo, letra e exceções do dia),
'        como Tabela do Excel, com filtro próprio. Letras do relatório: P =
'        entrada/saída da jornada, G = saída/volta da refeição, F = saída/volta
'        da pausa térmica (conferido: as pausas de 20 min e os períodos acima de
'        1h40 recontados a partir das batidas batem com o "Total do
'        Colaborador"). É a base para as vistas de horário, dia e turno.
'   A linha em branco depois da tabela 1 é proposital: é ela que marca o fim
'   da tabela de colaboradores para o GerarCSVPonto (GerarResumoPausasTermicas),
'   que lê só a tabela 1. Os totais NÃO são recalculados: vêm prontos do
'   relatório (são o número oficial). No fim, a aba é renomeada para
'   "Pausas Térmicas <mês do fim do período>".
'
' HISTÓRICO: uma versão antiga já tinha uma tabela por marcação; foi tirada
'   (só os totais eram usados) e voltou em 18/10, agora com uma linha por
'   batida, para o painel poder mostrar em que horário, dia e turno os
'   problemas acontecem.
'
' Rode este macro numa CÓPIA da planilha antes de usar em produção: ele
' reescreve a aba ativa por completo e o que uma macro faz NÃO volta com Ctrl+Z.
' ============================================================================
' COMO SE LIGA AO RESTO DO PROJETO
'   HRES114 (Senior) -> esta macro -> aba "Pausas Térmicas" na planilha de
'   Tratamento -> GerarAbaCSV (GerarCSVPonto) acha a tabela pela linha que tem
'   "Matrícula" + "Pausa corretas" e gera uma linha "Resumo Pausas Térmicas" por
'   colaborador (colunas pausas_corretas ... pausas_marcacoes_impares do CSV) ->
'   o painel mostra na seção "Pausas térmicas" e no card do colaborador.
'   Os nomes do cabeçalho (subHeaders, mais abaixo) são procurados pelo
'   GerarCSVPonto: mudar um nome aqui exige mudar lá.
'   Não depende de outros módulos (a única rotina de apoio, FormatarCabecalho,
'   está no fim deste arquivo).
' ============================================================================

' Macro principal: lê o relatório bruto de cima a baixo guardando os totais e
' depois reescreve a aba só com as duas tabelas.
Sub FormatarPausasTermicas()

    Dim ws As Worksheet
    Set ws = ActiveSheet

    ' lastRow = última linha com algo na coluna A. Aqui End(xlUp) é seguro porque o
    ' relatório recém-exportado não tem filtro (nas abas de Tratamento o projeto usa
    ' UltimaLinhaPreenchida por causa dos filtros).
    Dim lastRow As Long
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    If lastRow < 2 Then
        Aviso "Nenhum dado encontrado na aba ativa.", vbExclamation, "Formatar Pausas Térmicas"
        Exit Sub
    End If

    ' Confirmação antes de apagar: o usuário pode desistir.
    If Aviso("Este macro vai REESCREVER completamente a aba """ & ws.Name & _
              """, deixando os totais por colaborador, o total geral e as batidas dia a dia." & vbCrLf & vbCrLf & _
              "Recomendado: rode numa CÓPIA do arquivo." & vbCrLf & vbCrLf & _
              "Continuar?", vbYesNo + vbQuestion, "Formatar Pausas Térmicas") = vbNo Then
        Exit Sub
    End If

    ' Desliga atualização de tela e cálculo automático durante a macro (mais
    ' rápido); ambos são religados no fim. A barra de status mostra o andamento.
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.StatusBar = "Lendo dados originais..."

    ' ---------- Totais gerais originais (bloco "Total Geral" do relatório) ----------
    Dim geralPausaCorretas As Variant, geralPausaMenor As Variant, geralPausaMaior As Variant
    Dim geralTrabCorreto As Variant, geralTrabMaior As Variant, geralTrabMenor As Variant
    Dim geralMarcacoesImpares As Variant
    Dim temGeral As Boolean

    ' ---------- Totais por colaborador (um item por colaborador, bloco "Total do Colaborador") ----------
    Dim totColMatricula As Collection, totColNome As Collection, totColCargo As Collection
    Dim totColVals As Collection    ' cada item é um array(1 to 7)
    Set totColMatricula = New Collection
    Set totColNome = New Collection
    Set totColCargo = New Collection
    Set totColVals = New Collection

    ' Batidas de cada dia (18/10, pedido do RH): uma linha por batida, para a
    ' terceira tabela da aba (GuardarBatidasDoDia / EscreverTabelaBatidas).
    Dim batidas As Collection
    Set batidas = New Collection

    ' Contexto do colaborador "atual" enquanto varremos o relatório de cima a baixo: atualizado
    ' sempre que passamos pela linha de identificação do colaborador, e usado quando encontramos o
    ' bloco "Total do Colaborador" logo depois (esse bloco não repete matrícula/nome/cargo, então
    ' precisamos ter guardado isso momentos antes).
    Dim curMatricula As String, curNome As String, curCargo As String

    ' ============================================================
    ' Varredura do relatório bruto, linha a linha.
    ' Este relatório é um "dump" de impressão, não uma tabela: cada tipo de linha (marcação de dia,
    ' identificação do colaborador, subtotal do colaborador, total geral) tem um formato de
    ' texto/posição próprio, reconhecido aqui pelo conteúdo das colunas A/B/F. Linhas que não
    ' interessam para o resultado final (cabeçalho de página, legenda, "Soma dos itens", marcação
    ' dia a dia, rodapé) são simplesmente puladas (ramo Else, ao final) — não precisamos extrair nada
    ' delas, já que a saída deste macro não as usa.
    ' ============================================================
    Dim r As Long
    r = 1
    Do While r <= lastRow

        Dim a1 As Variant, b1 As Variant, f1 As Variant
        a1 = ws.Cells(r, 1).Value
        b1 = ws.Cells(r, 2).Value
        f1 = ws.Cells(r, 6).Value

        ' (05/11) A linha do dia é reconhecida pelo CONTEÚDO: data na coluna A (número de data,
        ' .Value2) e a 1ª batida (hora) ou nada na coluna B. Antes era pelo formato "DD/MM/YYYY"
        ' da célula, que o Excel PERDE quando o arquivo do sistema (com estilos defeituosos) é
        ' trazido pela GerarCSVsDaPasta: nenhuma batida era lida e só saía um mês.
        If EhLinhaDeDia(ws, r) Then
            ' ---------------- Linha de marcações do dia (uma data + as batidas de ponto daquele
            ' dia, nas colunas B, D, F...) ----------------
            ' Os totais (Pausa corretas/menor/maior, Trabalho correto/maior/menor, Marcações
            ' Ímpares) NÃO são recalculados a partir dessas batidas: vêm prontos no bloco "Total
            ' do Colaborador". As batidas são guardadas (18/10) só para a terceira tabela da aba,
            ' a de marcações dia a dia (GuardarBatidasDoDia).
            If curMatricula <> "" Then GuardarBatidasDoDia ws, r, curMatricula, curNome, batidas
            r = r + 1

        ElseIf IsNumeric(a1) And Not IsEmpty(a1) _
                And Len(Trim(CStr(b1))) > 0 And Not IsNumeric(b1) _
                And Len(Trim(CStr(f1))) > 0 Then
            ' ---------------- Linha de identificação do colaborador ----------------
            ' Formato: coluna A = matrícula (número, mas SEM o formato de data DD/MM/YYYY das
            ' linhas de marcação), coluna B = nome (texto), coluna F = cargo (texto). Guardamos os
            ' três em curMatricula/curNome/curCargo para rotular o bloco "Total do Colaborador"
            ' deste mesmo colaborador, que vem logo a seguir no relatório.
            curMatricula = CStr(CLng(a1))
            curNome = Trim(CStr(b1))
            curCargo = Trim(CStr(f1))
            r = r + 1

        ElseIf Trim(CStr(a1)) = "Total do Colaborador" Then
            ' ---------------- Subtotais do colaborador (já calculados pelo relatório original) ----
            ' Os 7 números ficam sempre 2 linhas abaixo do rótulo "Total do Colaborador", em
            ' colunas fixas (posições definidas pelo layout de impressão do relatório original) -
            ' por isso os deslocamentos "r + 2" e os números de coluna abaixo são fixos, não
            ' calculados por cabeçalho como em outras partes deste projeto.
            Dim valsCol(1 To 7) As Variant
            valsCol(1) = ws.Cells(r + 2, 1).Value  ' Pausa corretas
            valsCol(2) = ws.Cells(r + 2, 2).Value  ' Pausa menor 0:20
            valsCol(3) = ws.Cells(r + 2, 4).Value  ' Pausa maior 0:20
            valsCol(4) = ws.Cells(r + 2, 6).Value  ' Trabalho correto 1:40
            valsCol(5) = ws.Cells(r + 2, 8).Value  ' Trabalho maior 1:40
            valsCol(6) = ws.Cells(r + 2, 10).Value ' Trabalho menor 1:40
            valsCol(7) = ws.Cells(r + 2, 11).Value ' Marcações Ímpares

            totColMatricula.Add curMatricula
            totColNome.Add curNome
            totColCargo.Add curCargo
            totColVals.Add valsCol

            r = r + 3

        ElseIf Trim(CStr(a1)) = "Total" And Trim(CStr(b1)) = "Geral" Then
            ' ---------------- Total geral (soma de todos os colaboradores, já calculada) ----------
            ' Mesma lógica de deslocamento fixo "r + 2" do bloco "Total do Colaborador" acima, só
            ' que aqui é a única ocorrência do relatório inteiro (por isso a flag temGeral, usada
            ' na hora de escrever a saída: só escrevemos a seção "Total Geral" se ela existir).
            geralPausaCorretas = ws.Cells(r + 2, 1).Value
            geralPausaMenor = ws.Cells(r + 2, 2).Value
            geralPausaMaior = ws.Cells(r + 2, 4).Value
            geralTrabCorreto = ws.Cells(r + 2, 6).Value
            geralTrabMaior = ws.Cells(r + 2, 8).Value
            geralTrabMenor = ws.Cells(r + 2, 10).Value
            geralMarcacoesImpares = ws.Cells(r + 2, 11).Value
            temGeral = True
            r = r + 3

        Else
            ' ---------------- Qualquer outra linha: cabeçalho de página, legenda original do
            ' relatório, "Soma dos itens 1 e 3...", rodapé (linhas "HRES...") etc. ----------------
            ' Nenhuma dessas informações faz parte da saída deste macro (pedido do RH: "só as
            ' tabelas dos colaboradores e a tabela de total, mais nada"), então essas linhas são
            ' simplesmente ignoradas - avançamos uma linha e seguimos procurando os blocos que
            ' interessam.
            r = r + 1
        End If
    Loop

    ' Fim do período do relatório (linha "Período: dd/mm/aaaa até dd/mm/aaaa", no
    ' cabeçalho): diz de que mês é a aba. Lido ANTES de apagar o relatório; é a última
    ' data da linha do "Período:". 0 = não achou (a aba fica com o nome que tem).
    Dim fimPeriodo As Date, rp As Long, cp As Long, achouPeriodo As Boolean
    For rp = 1 To 15
        For cp = 1 To 15
            If Trim$(CStr(ws.Cells(rp, cp).Value)) = "Período:" Then achouPeriodo = True
        Next cp
        If achouPeriodo Then
            ' (05/11) pelo número da data (.Value2), com ou sem formato de data na célula
            Dim vp As Variant
            For cp = 1 To 20
                vp = ws.Cells(rp, cp).Value2
                If VarType(vp) = vbDouble Then
                    If vp > 36526 And vp < 73415 Then
                        If CDate(vp) > fimPeriodo Then fimPeriodo = CDate(vp)
                    End If
                End If
            Next cp
            Exit For
        End If
    Next rp

    ' Relatório de VÁRIOS meses num arquivo só (31/10): o "Total do Colaborador" do
    ' sistema vem somado em todos os meses, mas o painel usa um mês por aba. Então,
    ' quando as batidas passam de um ciclo (28 a 27), a macro faz uma aba por ciclo,
    ' com os totais RECONTADOS das batidas daquele ciclo (RecontarPorCiclo).
    Dim ciclos As Collection, ti As Long
    Set ciclos = CiclosDasBatidas(batidas)

    If ciclos.Count > 1 Then
        Dim cargoPorMat As Object, ic As Long, fimCiclo As Date, wsMes As Worksheet
        Dim msgMeses As String, nomeMes As String
        Set cargoPorMat = CreateObject("Scripting.Dictionary")
        For ti = 1 To totColMatricula.Count
            If Not cargoPorMat.Exists(totColMatricula(ti)) Then cargoPorMat.Add totColMatricula(ti), totColCargo(ti)
        Next ti
        For ic = 1 To ciclos.Count
            fimCiclo = ciclos(ic)
            Application.StatusBar = "Montando o mês " & NomeDoMes(Month(fimCiclo)) & "..."
            If ic = ciclos.Count Then
                Set wsMes = ws     ' o mês mais recente fica na própria aba do relatório
            Else
                Set wsMes = ws.Parent.Worksheets.Add(Before:=ws)
            End If
            EscreverMesRecontado wsMes, batidas, fimCiclo, cargoPorMat
            nomeMes = "Pausas Térmicas " & NomeDoMes(Month(fimCiclo))
            If wsMes.Name <> nomeMes Then
                If AbaExiste(ws.Parent, nomeMes) Then
                    msgMeses = msgMeses & vbCrLf & "- " & NomeDoMes(Month(fimCiclo)) & ": já existia uma aba """ & nomeMes & _
                        """; esta ficou como """ & wsMes.Name & """."
                Else
                    wsMes.Name = nomeMes
                    msgMeses = msgMeses & vbCrLf & "- aba """ & nomeMes & """ (ciclo até " & Format$(fimCiclo, "dd/mm/yyyy") & ")"
                End If
            End If
        Next ic
        Application.ScreenUpdating = True
        Application.Calculation = xlCalculationAutomatic
        Application.StatusBar = False
        Aviso "Formatação concluída. O relatório tinha " & ciclos.Count & " meses (ciclos de 28 a 27): " & _
              "foi separado em uma aba por mês, com os totais recontados a partir das batidas de cada mês " & _
              "(o total do sistema vinha somado nos " & ciclos.Count & " meses)." & msgMeses, _
              vbInformation, "Formatar Pausas Térmicas"
        Exit Sub
    End If

    Application.StatusBar = "Reescrevendo a planilha..."
    EscreverAbaPausas ws, totColMatricula, totColNome, totColCargo, totColVals, temGeral, _
        Array(geralPausaCorretas, geralPausaMenor, geralPausaMaior, geralTrabCorreto, geralTrabMaior, geralTrabMenor, geralMarcacoesImpares), batidas

    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    Application.StatusBar = False

    ' Renomeia a aba pelo mês do fim do período ("Pausas Térmicas Setembro"), para
    ' o GerarCSVPonto saber de que mês é cada aba quando houver mais de uma (T9).
    ' Se o período não for achado, ou já existir aba com esse nome, mantém o nome.
    Dim msgNome As String, novoNome As String
    If fimPeriodo > 0 Then
        novoNome = "Pausas Térmicas " & NomeDoMes(Month(fimPeriodo))
        If ws.Name <> novoNome Then
            If AbaExiste(ws.Parent, novoNome) Then
                msgNome = "Já existe uma aba """ & novoNome & """: esta ficou com o nome """ & ws.Name & """."
            Else
                ws.Name = novoNome
                msgNome = "Aba renomeada para """ & novoNome & """ (período até " & Format$(fimPeriodo, "dd/mm/yyyy") & ")."
            End If
        End If
    Else
        msgNome = "Período do relatório não encontrado: a aba ficou com o nome """ & ws.Name & """."
    End If

    Aviso "Formatação concluída." & vbCrLf & _
           totColMatricula.Count & " colaborador(es) na tabela de colaboradores." & vbCrLf & _
           IIf(temGeral, "Total Geral incluído.", "Bloco ""Total Geral"" não encontrado no relatório original.") & vbCrLf & _
           batidas.Count & " batida(s) na tabela de marcações (embaixo do Total Geral)." & _
           IIf(msgNome <> "", vbCrLf & msgNome, ""), _
           vbInformation, "Formatar Pausas Térmicas"

End Sub


' Visual padrão das linhas de cabeçalho das duas tabelas (negrito, fundo cinza, borda).
' Lê uma linha de dia do relatório: depois da data (coluna A) vêm pares
' "horário + letra" (P = entrada/saída da jornada, G = saída/volta da
' refeição, F = saída/volta da pausa térmica) e, no fim, as exceções do dia
' entre parênteses ("(1) (3)", ver a legenda do relatório). Guarda uma linha
' por batida em "batidas": matrícula, nome, data, hora (fração do dia), nº da
' batida no dia, letra e exceções. Batida depois da meia-noite continua na
' data da linha (é o mesmo turno).
' (05/11) Linha de dia do relatório: coluna A = data (número de data entre 2000 e 2100) e
' coluna B = a 1ª batida (hora, número menor que 1) ou vazia. A linha da pessoa tem o nome
' (texto) na coluna B; a do "Total do Colaborador", números pequenos na coluna A.
Private Function EhLinhaDeDia(ws As Worksheet, ByVal r As Long) As Boolean
    Dim a As Variant, b As Variant
    a = ws.Cells(r, 1).Value2
    If VarType(a) <> vbDouble Then Exit Function
    If a <= 36526 Or a >= 73415 Then Exit Function
    b = ws.Cells(r, 2).Value2
    If IsEmpty(b) Then
        EhLinhaDeDia = True
    ElseIf VarType(b) = vbDouble Then
        EhLinhaDeDia = (b >= 0 And b < 1)
    End If
End Function

Private Sub GuardarBatidasDoDia(ws As Worksheet, ByVal r As Long, ByVal mat As String, ByVal nome As String, batidas As Collection)
    Dim vals As Variant, c As Long, seq As Long, v As Variant, letra As String, excecoes As String
    vals = ws.Range(ws.Cells(r, 1), ws.Cells(r, 45)).Value2
    For c = 2 To 45
        If VarType(vals(1, c)) = vbString Then
            If Left$(Trim$(vals(1, c)), 1) = "(" Then excecoes = Trim$(vals(1, c))
        End If
    Next c
    For c = 2 To 44
        v = vals(1, c)
        If VarType(v) = vbDouble Then
            letra = Trim$(CStr(vals(1, c + 1)))
            If Len(letra) = 1 Then
                seq = seq + 1
                batidas.Add Array(mat, nome, vals(1, 1), v, seq, letra, excecoes)
            End If
        End If
    Next c
End Sub

' Escreve a tabela de batidas a partir da linha "linhaTitulo" (título, depois
' cabeçalho e dados) e a transforma numa Tabela do Excel, que tem filtro
' próprio (o filtro da aba continua na tabela de totais). Os dados vão de uma
' vez, por uma matriz, porque são milhares de linhas.
Private Sub EscreverTabelaBatidas(ws As Worksheet, ByVal linhaTitulo As Long, batidas As Collection)
    Dim cab As Variant, n As Long, i As Long, b As Variant, dados() As Variant, linhaCab As Long
    cab = Array("Matrícula", "Colaborador", "Data", "Dia da semana", "Hora", "Nº da batida no dia", _
                "Tipo da batida", "Letra no relatório", "Exceções do dia")
    n = batidas.Count
    ReDim dados(1 To n, 1 To 9)
    ' For Each (e não batidas(i)): pegar item de Collection pelo número percorre a
    ' lista desde o começo a cada vez, o que ficaria lento com milhares de batidas.
    i = 0
    For Each b In batidas
        i = i + 1
        dados(i, 1) = CLng(b(0))
        dados(i, 2) = b(1)
        dados(i, 3) = b(2)
        dados(i, 4) = Choose(Weekday(b(2)), "Dom", "Seg", "Ter", "Qua", "Qui", "Sex", "Sáb")
        dados(i, 5) = b(3) - Int(b(3))
        dados(i, 6) = b(4)
        dados(i, 7) = TipoDaBatida(b(5))
        dados(i, 8) = b(5)
        dados(i, 9) = TraduzirExcecoes(b(6))
    Next b

    ws.Cells(linhaTitulo, 1).Value = "Marcações do período (uma linha por batida)"
    ws.Cells(linhaTitulo, 1).Font.Bold = True
    linhaCab = linhaTitulo + 1
    ws.Range(ws.Cells(linhaCab, 1), ws.Cells(linhaCab, 9)).NumberFormat = "@"
    ws.Range(ws.Cells(linhaCab, 1), ws.Cells(linhaCab, 9)).Value = cab
    ws.Range(ws.Cells(linhaCab + 1, 1), ws.Cells(linhaCab + n, 9)).Value = dados
    ws.Range(ws.Cells(linhaCab + 1, 3), ws.Cells(linhaCab + n, 3)).NumberFormat = "dd/mm/yyyy"
    ws.Range(ws.Cells(linhaCab + 1, 5), ws.Cells(linhaCab + n, 5)).NumberFormat = "hh:mm"

    Dim lo As ListObject
    Set lo = ws.ListObjects.Add(xlSrcRange, ws.Range(ws.Cells(linhaCab, 1), ws.Cells(linhaCab + n, 9)), , xlYes)
    lo.TableStyle = "TableStyleLight1"
End Sub

' Letra da batida no relatório -> texto (P, G, F; outra letra fica como veio).
Private Function TipoDaBatida(ByVal letra As String) As String
    Select Case UCase$(letra)
        Case "P": TipoDaBatida = "Entrada ou saída da jornada"
        Case "G": TipoDaBatida = "Saída ou volta da refeição"
        Case "F": TipoDaBatida = "Saída ou volta da pausa térmica"
        Case Else: TipoDaBatida = "Outra (" & letra & ")"
    End Select
End Function

' "(1) (3)" -> "Pausa menor que 0:20; Trabalho maior que 1:40" (legenda do relatório).
Private Function TraduzirExcecoes(ByVal codigos As String) As String
    Dim nomes As Variant, i As Long, t As String
    nomes = Array("", "Pausa menor que 0:20", "Pausa maior que 0:20", "Trabalho maior que 1:40", _
                  "Trabalho menor que 1:40", "Marcações ímpares", "Pausa correta")
    For i = 1 To 6
        If InStr(1, codigos, "(" & i & ")") > 0 Then t = t & IIf(t = "", "", "; ") & nomes(i)
    Next i
    TraduzirExcecoes = t
End Function

' Diz se a pasta já tem uma aba com esse nome (sem ligar para maiúsculas).
Private Function AbaExiste(wb As Workbook, ByVal nome As String) As Boolean
    Dim sh As Object
    For Each sh In wb.Sheets
        If LCase$(sh.Name) = LCase$(nome) Then AbaExiste = True: Exit Function
    Next sh
    AbaExiste = False
End Function

    With rng
        .Font.Bold = True
        .Interior.Color = RGB(217, 217, 217)
        .Borders.LineStyle = xlContinuous
        .HorizontalAlignment = xlCenter
        .VerticalAlignment = xlCenter
        .WrapText = True
    End With
End Sub

' ----------------------------------------------------------------------------
' Escreve numa aba (que é limpa antes) as três tabelas: colaboradores (linha 1),
' Total Geral e as batidas. É o mesmo desenho de sempre; virou rotina própria em
' 31/10 para servir também às abas de cada mês (EscreverMesRecontado).
'   geral = Array dos 7 números do Total Geral (pausas corretas ... marcações ímpares).
' ----------------------------------------------------------------------------
Private Sub EscreverAbaPausas(ws As Worksheet, mats As Collection, nomes As Collection, cargos As Collection, _
                              vals As Collection, ByVal temGeral As Boolean, geral As Variant, batidas As Collection)
    If ws.AutoFilterMode Then ws.AutoFilterMode = False
    Do While ws.ListObjects.Count > 0     ' tabela de batidas de uma formatação anterior
        ws.ListObjects(1).Unlist
    Loop
    ws.Cells.Clear
    ws.Activate
    ActiveWindow.FreezePanes = False

    Dim subHeaders As Variant
    subHeaders = Array("Matrícula", "Colaborador", "Cargo", "Pausa corretas", _
                        "Pausa menor 0:20", "Pausa maior 0:20", "Trabalho correto 1:40", _
                        "Trabalho maior 1:40", "Trabalho menor 1:40", "Marcações Ímpares")

    Dim writeRow As Long, shRow As Long, hcol As Long, ti As Long, k As Long, vv As Variant
    shRow = 1
    For hcol = 1 To 10
        ws.Cells(shRow, hcol).Value = subHeaders(hcol - 1)
    Next hcol
    FormatarCabecalho ws.Range(ws.Cells(shRow, 1), ws.Cells(shRow, 10))
    writeRow = shRow + 1

    For ti = 1 To mats.Count
        vv = vals(ti)
        ws.Cells(writeRow, 1).Value = mats(ti)
        ws.Cells(writeRow, 2).Value = nomes(ti)
        ws.Cells(writeRow, 3).Value = cargos(ti)
        For k = 1 To 7
            ws.Cells(writeRow, 3 + k).Value = vv(k)
        Next k
        writeRow = writeRow + 1
    Next ti

    If mats.Count > 0 Then
        With ws.Range(ws.Cells(shRow, 1), ws.Cells(writeRow - 1, 10))
            .Borders.LineStyle = xlContinuous
            .Borders.Weight = xlThin
        End With
        ws.Range(ws.Cells(shRow, 1), ws.Cells(writeRow - 1, 10)).AutoFilter
        ws.Rows(shRow + 1).Select
        ActiveWindow.FreezePanes = True
    End If

    ' Total Geral, depois de UMA linha em branco (ela marca o fim da tabela 1
    ' para o GerarCSVPonto)
    If temGeral Then
        writeRow = writeRow + 1
        With ws.Range(ws.Cells(writeRow, 1), ws.Cells(writeRow, 3))
            .Merge
            .Value = "Total Geral"
        End With
        For hcol = 4 To 10
            ws.Cells(writeRow, hcol).Value = subHeaders(hcol - 1)
        Next hcol
        FormatarCabecalho ws.Range(ws.Cells(writeRow, 1), ws.Cells(writeRow, 10))
        writeRow = writeRow + 1
        With ws.Range(ws.Cells(writeRow, 1), ws.Cells(writeRow, 3))
            .Merge
            .Value = "Todos os colaboradores"
        End With
        For k = 0 To 6
            ws.Cells(writeRow, 4 + k).Value = geral(k)
        Next k
        ws.Range(ws.Cells(writeRow, 1), ws.Cells(writeRow, 10)).Font.Bold = True
        With ws.Range(ws.Cells(writeRow - 1, 1), ws.Cells(writeRow, 10))
            .Borders.LineStyle = xlContinuous
            .Borders.Weight = xlThin
        End With
    End If

    ' Batidas, duas linhas em branco depois do Total Geral
    If batidas.Count > 0 Then EscreverTabelaBatidas ws, writeRow + 3, batidas

    ws.Cells.Font.Name = "Calibri"
    ws.Cells.Font.Size = 10
    ws.Columns("A").ColumnWidth = 12
    ws.Columns("B").ColumnWidth = 32
    ws.Columns("C").ColumnWidth = 28
    ws.Columns("D").ColumnWidth = 14
    ws.Columns("E").ColumnWidth = 14
    ws.Columns("F").ColumnWidth = 14
    ws.Columns("G").ColumnWidth = 18
    ws.Columns("H").ColumnWidth = 16
    ws.Columns("I").ColumnWidth = 16
    ws.Columns("J").ColumnWidth = 16
    ws.Range("D:J").HorizontalAlignment = xlCenter
    ws.Cells(1, 1).Select
End Sub

' Dia 27 que fecha o ciclo (28 a 27) de uma data: 28/09 -> 27/10; 27/10 -> 27/10.
Private Function FimDoCiclo(ByVal d As Date) As Date
    If Day(d) >= 28 Then
        FimDoCiclo = DateSerial(Year(d), Month(d) + 1, 27)
    Else
        FimDoCiclo = DateSerial(Year(d), Month(d), 27)
    End If
End Function

' Os ciclos que aparecem nas batidas, do mais antigo ao mais recente (cada item =
' o dia 27 que fecha o ciclo). A data da batida é a do turno (coluna A do relatório).
Private Function CiclosDasBatidas(batidas As Collection) As Collection
    Dim vistos As Object, b As Variant, f As Date, chaves As Variant, i As Long, j As Long, tmp As Variant
    Set vistos = CreateObject("Scripting.Dictionary")
    For Each b In batidas
        f = FimDoCiclo(CDate(b(2)))
        If Not vistos.Exists(CLng(f)) Then vistos.Add CLng(f), f
    Next b
    Set CiclosDasBatidas = New Collection
    If vistos.Count = 0 Then Exit Function
    chaves = vistos.Keys
    For i = 0 To UBound(chaves) - 1
        For j = i + 1 To UBound(chaves)
            If chaves(j) < chaves(i) Then tmp = chaves(i): chaves(i) = chaves(j): chaves(j) = tmp
        Next j
    Next i
    For i = 0 To UBound(chaves)
        CiclosDasBatidas.Add vistos(chaves(i))
    Next i
End Function

' Monta numa aba o mês (ciclo) que termina em fimCiclo, com os totais RECONTADOS
' das batidas daquele ciclo. Regras (as que bateram com o total do sistema nos 3
' meses baixados em 31/10): em cada dia, as batidas em ordem alternam trabalho e
' pausa (1ª->2ª trabalho, 2ª->3ª pausa...); pausa com as duas batidas F = pausa
' térmica: 20 min ou mais = correta, menos = menor ("maior" o sistema não deu
' nenhuma); trecho de trabalho = 1h40 certinho correto, mais = maior, menos =
' menor; dia com número ímpar de batidas = 1 marcação ímpar. Hora que "volta"
' (turno que passa da meia-noite) soma 24h. Pausas conferem com o sistema; os
' trechos de trabalho ficam até 1,5% abaixo (dias com batida faltando).
Private Sub EscreverMesRecontado(ws As Worksheet, batidas As Collection, ByVal fimCiclo As Date, cargoPorMat As Object)
    Dim doMes As Collection, b As Variant
    Set doMes = New Collection
    For Each b In batidas
        If FimDoCiclo(CDate(b(2))) = fimCiclo Then doMes.Add b
    Next b

    ' totais por matrícula, na ordem em que as pessoas aparecem
    Dim tot As Object, ordem As Collection, nomePor As Object
    Set tot = CreateObject("Scripting.Dictionary")
    Set nomePor = CreateObject("Scripting.Dictionary")
    Set ordem = New Collection
    Dim minutos() As Long, letras() As String, n As Long, chaveDia As String, chaveAnt As String
    Dim matAnt As String, i As Long
    ReDim minutos(1 To 100): ReDim letras(1 To 100)
    n = 0
    For Each b In doMes
        chaveDia = b(0) & "|" & CLng(b(2))
        If chaveDia <> chaveAnt And n > 0 Then
            ContarDia tot, matAnt, minutos, letras, n
            n = 0
        End If
        If Not tot.Exists(CStr(b(0))) Then
            Dim zeros(1 To 7) As Long
            tot.Add CStr(b(0)), zeros
            nomePor.Add CStr(b(0)), b(1)
            ordem.Add CStr(b(0))
        End If
        If n < 100 Then
            n = n + 1
            minutos(n) = CLng(Round((CDbl(b(3)) - Int(CDbl(b(3)))) * 1440, 0))
            letras(n) = UCase$(CStr(b(5)))
        End If
        chaveAnt = chaveDia
        matAnt = CStr(b(0))
    Next b
    If n > 0 Then ContarDia tot, matAnt, minutos, letras, n

    Dim mats As New Collection, nomes As New Collection, cargos As New Collection, vals As New Collection
    Dim m As Variant, v As Variant, geral(0 To 6) As Long, k As Long
    For Each m In ordem
        v = tot(m)
        Dim linhaVals(1 To 7) As Variant
        For k = 1 To 7
            linhaVals(k) = v(k)
            geral(k - 1) = geral(k - 1) + v(k)
        Next k
        mats.Add m
        nomes.Add nomePor(m)
        If cargoPorMat.Exists(m) Then cargos.Add cargoPorMat(m) Else cargos.Add ""
        vals.Add linhaVals
    Next m
    EscreverAbaPausas ws, mats, nomes, cargos, vals, True, _
        Array(geral(0), geral(1), geral(2), geral(3), geral(4), geral(5), geral(6)), doMes
End Sub

' Soma num dia de uma pessoa (minutos/letras das n batidas, em ordem) os 7 totais.
Private Sub ContarDia(tot As Object, ByVal mat As String, minutos() As Long, letras() As String, ByVal n As Long)
    Dim v As Variant, j As Long, base As Long, t() As Long, dur As Long
    ReDim t(1 To n)
    base = 0
    For j = 1 To n
        t(j) = minutos(j) + base
        If j > 1 Then
            If t(j) < t(j - 1) Then base = base + 1440: t(j) = t(j) + 1440
        End If
    Next j
    v = tot(mat)
    If n Mod 2 = 1 Then v(7) = v(7) + 1
    For j = 1 To n - 1
        dur = t(j + 1) - t(j)
        If j Mod 2 = 1 Then
            ' trecho de trabalho (1ª->2ª, 3ª->4ª...)
            If dur = 100 Then
                v(4) = v(4) + 1
            ElseIf dur > 100 Then
                v(5) = v(5) + 1
            Else
                v(6) = v(6) + 1
            End If
        ElseIf letras(j) = "F" And letras(j + 1) = "F" Then
            ' pausa térmica
            If dur >= 20 Then v(1) = v(1) + 1 Else v(2) = v(2) + 1
        End If
    Next j
    tot(mat) = v
End Sub

' ----------------------------------------------------------------------------
' Modo automático (31/10). FormatarPausasTermicas_Auto é chamada pela macro
' PrepararAbasDoMes com a aba do relatório ativa: roda a formatação sem nenhuma
' caixa de mensagem e devolve, em texto, o que teria aparecido nelas.
' Aviso substitui o MsgBox: fora do modo automático é o MsgBox de sempre; no
' automático guarda o texto (com "ERRO: " na frente quando o aviso é de erro) e
' responde "Sim" às perguntas (ex.: "vai reescrever a aba, continuar?").
' ----------------------------------------------------------------------------
Public Function FormatarPausasTermicas_Auto() As String
    modoAuto = True: msgAuto = ""
    FormatarPausasTermicas
    modoAuto = False
    FormatarPausasTermicas_Auto = msgAuto
End Function

Private Function Aviso(ByVal texto As String, Optional ByVal botoes As Long = vbInformation, Optional ByVal titulo As String = "") As VbMsgBoxResult
    If modoAuto Then
        If (botoes And 16) = 16 Then texto = "ERRO: " & texto
        msgAuto = msgAuto & IIf(msgAuto = "", "", " ") & Replace(Replace(texto, vbCrLf, " "), vbLf, " ")
        Aviso = vbYes
    Else
        Aviso = MsgBox(texto, botoes, titulo)
    End If
End Function
